cmake_minimum_required(VERSION 3.16)
project(twrpv2 C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(TW_X11 "Build the X11 simulator window (desktop only)" ON)
option(TW_STATIC "Link fully static (for the recovery ramdisk)" OFF)

file(GLOB_RECURSE TW_SOURCES CONFIGURE_DEPENDS src/*.cpp)
file(GLOB TW_XZ_SOURCES third_party/xz/*.c)
add_executable(twrpv2 ${TW_SOURCES} third_party/miniz/miniz.c ${TW_XZ_SOURCES})
target_include_directories(twrpv2 PRIVATE src third_party/stb third_party/miniz third_party/xz)
string(TIMESTAMP TW_BUILD_DATE "%Y-%m-%d")
# 64-bit file offsets: GSIs and backups are often larger than 2 GiB, also on 32-bit ARM.
add_compile_definitions(_FILE_OFFSET_BITS=64 _LARGEFILE64_SOURCE)
target_compile_definitions(twrpv2 PRIVATE MINIZ_NO_ZLIB_COMPATIBLE_NAMES TW_BUILD_DATE="${TW_BUILD_DATE}")
target_compile_options(twrpv2 PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall -Wextra -Wno-unused-parameter -Wno-missing-field-initializers>)
find_package(Threads REQUIRED)
target_link_libraries(twrpv2 PRIVATE Threads::Threads)

if(TW_X11 AND NOT ANDROID)
  find_package(X11)
  if(X11_FOUND)
    target_compile_definitions(twrpv2 PRIVATE TW_HAVE_X11)
    target_link_libraries(twrpv2 PRIVATE ${X11_LIBRARIES})
    target_include_directories(twrpv2 PRIVATE ${X11_INCLUDE_DIR})
    message(STATUS "TWRP v2: X11 simulator enabled")
  endif()
endif()

if(TW_STATIC)
  target_link_options(twrpv2 PRIVATE -static -s)
endif()

# Host self-tests (installer protocol, sparse images, backup/restore round trip)
if(NOT CMAKE_CROSSCOMPILING)
  set(TW_TEST_SOURCES ${TW_SOURCES})
  list(FILTER TW_TEST_SOURCES EXCLUDE REGEX ".*/src/main\\.cpp$")
  list(FILTER TW_TEST_SOURCES EXCLUDE REGEX ".*/src/(ui|gfx)/.*")
  list(FILTER TW_TEST_SOURCES EXCLUDE REGEX ".*/display_.*\\.cpp$")
  list(FILTER TW_TEST_SOURCES EXCLUDE REGEX ".*/input_evdev\\.cpp$")
  add_executable(twrpv2_selftest tests/selftest.cpp ${TW_TEST_SOURCES} third_party/miniz/miniz.c ${TW_XZ_SOURCES})
  target_include_directories(twrpv2_selftest PRIVATE src third_party/stb third_party/miniz third_party/xz)
  target_compile_definitions(twrpv2_selftest PRIVATE MINIZ_NO_ZLIB_COMPATIBLE_NAMES)
  target_link_libraries(twrpv2_selftest PRIVATE Threads::Threads)
  enable_testing()
  add_test(NAME selftest COMMAND twrpv2_selftest)
endif()

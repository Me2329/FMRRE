cmake_minimum_required(VERSION 3.16)
project(fmrre C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(TW_X11 "Build the X11 simulator window (desktop only)" ON)
option(TW_STATIC "Link fully static (for the recovery ramdisk)" OFF)

file(GLOB_RECURSE TW_SOURCES CONFIGURE_DEPENDS src/*.cpp)
file(GLOB TW_XZ_SOURCES third_party/xz/*.c)
add_executable(fmrre ${TW_SOURCES} third_party/miniz/miniz.c ${TW_XZ_SOURCES})
target_include_directories(fmrre PRIVATE src third_party/stb third_party/miniz third_party/xz)
string(TIMESTAMP TW_BUILD_DATE "%Y-%m-%d")
# 64-bit file offsets: GSIs and backups are often larger than 2 GiB, also on 32-bit ARM.
add_compile_definitions(_FILE_OFFSET_BITS=64 _LARGEFILE64_SOURCE)
target_compile_definitions(fmrre PRIVATE MINIZ_NO_ZLIB_COMPATIBLE_NAMES TW_BUILD_DATE="${TW_BUILD_DATE}")
target_compile_options(fmrre PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall -Wextra -Wno-unused-parameter -Wno-missing-field-initializers>)
find_package(Threads REQUIRED)
target_link_libraries(fmrre PRIVATE Threads::Threads)

if(TW_X11 AND NOT ANDROID)
  find_package(X11)
  if(X11_FOUND)
    target_compile_definitions(fmrre PRIVATE TW_HAVE_X11)
    target_link_libraries(fmrre PRIVATE ${X11_LIBRARIES})
    target_include_directories(fmrre PRIVATE ${X11_INCLUDE_DIR})
    message(STATUS "FMRRE: X11 simulator enabled")
  endif()
endif()

if(TW_STATIC)
  target_link_options(fmrre PRIVATE -static -s)
endif()

# Self-tests (installer protocol, images, super metadata, backup/restore round trip). Cross builds can
# include them too (-DTW_SELFTEST=ON) to run under qemu-user: scripts/test_archs.sh does that.
if(CMAKE_CROSSCOMPILING)
  option(TW_SELFTEST "Build the self-test binary (run it with qemu-user)" OFF)
else()
  option(TW_SELFTEST "Build the self-test binary" ON)
endif()
if(TW_SELFTEST)
  set(TW_TEST_SOURCES ${TW_SOURCES})
  list(FILTER TW_TEST_SOURCES EXCLUDE REGEX ".*/src/main\\.cpp$")
  list(FILTER TW_TEST_SOURCES EXCLUDE REGEX ".*/src/(ui|gfx)/.*")
  list(FILTER TW_TEST_SOURCES EXCLUDE REGEX ".*/display_.*\\.cpp$")
  add_executable(fmrre_selftest tests/selftest.cpp ${TW_TEST_SOURCES} third_party/miniz/miniz.c ${TW_XZ_SOURCES})
  target_include_directories(fmrre_selftest PRIVATE src third_party/stb third_party/miniz third_party/xz)
  target_compile_definitions(fmrre_selftest PRIVATE MINIZ_NO_ZLIB_COMPATIBLE_NAMES)
  target_link_libraries(fmrre_selftest PRIVATE Threads::Threads)
  enable_testing()
  add_test(NAME selftest COMMAND fmrre_selftest)
endif()
